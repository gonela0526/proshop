    previous = self.session_service.get_session(session_id).get("last_response")

    if not previous:
        return {
            "status": "error",
            "message": "No previous result available to transform.",
            "session_id": session_id
        }

    transformed = self.llm.transform_response(previous, query)

    return {
        "status": "success",
        "intent": "transform_previous",
        "data": {
            "result": transformed
        },
        "session_id": session_id
    }

def transform_response(self, previous_response: dict, user_query: str):

    prompt = f"""
You are an intelligent assistant working inside an enterprise exchange intelligence system.

You are given:
1) The previous system response (JSON)
2) A new user request that refers to that response

Your job:
- Transform the previous response according to the user's request.
- You must NOT invent new data.
- You must ONLY use the provided previous response.
- If the user asks for timezone conversion, convert properly.
- If the user asks for explanation, summarize clearly.
- If the request is unclear, respond with a clarification message.

Return ONLY plain text.
Do NOT return JSON.
Do NOT wrap in code blocks.

Previous Response:
{previous_response}

User Request:
{user_query}

Final Answer:
"""

    return self._call_llm(prompt)






def classify_intent(self, query: str):

    prompt = f"""
You are an intent classifier.

Determine if the user is asking to:
- transform previous result
- convert timezone
- summarize previous answer
- explain previous answer
- reformat previous answer

If yes, return:
{{ "intent": "transform_previous" }}

If not, return:
{{ "intent": null }}

User Query:
{query}
"""

    response = self._call_llm(prompt)
    return self._safe_json_parse(response)
