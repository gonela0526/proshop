import json
import re

def _extract_json(self, text: str):
    try:
        # Remove markdown blocks if present
        text = re.sub(r"```json", "", text)
        text = re.sub(r"```", "", text)

        # Extract first {...}
        match = re.search(r"\{.*\}", text, re.DOTALL)
        if not match:
            return {"intent": "unknown"}

        return json.loads(match.group())

    except Exception:
        return {"intent": "unknown"}
