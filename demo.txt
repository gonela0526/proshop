package com.agosense.symphony.process.visteon.jira2jira_gen;

public class StatusExchange {

    private static final Logger LOGGER = LoggerFactory.getLogger(com.agosense.symphony.process.visteon.jira2jira_gen.StatusExchange.class);

    JiraOSGiServiceI jira;

    String statusFilePath;
    EntityType sourceEntity;
    EntityType targetEntity;
    Jira2JiraCfg cfg;

    public StatusExchange(String statusFilePath, EntityType sourceEntity, EntityType targetEntity,Jira2JiraCfg cfg,JiraOSGiServiceI jira) {
        this.statusFilePath = statusFilePath;
        this.sourceEntity = sourceEntity;
        this.targetEntity = targetEntity;
        this.cfg = cfg;
        this.jira = jira;
    }

    public void execute() throws IOException,JiraException, ParseException
    {
        LOGGER.info("inside the Status Exchange class in execute method");
        Map<String,List<StatusMapObject>> sourceCondnMap = returnStatusMapObjectList();

        LOGGER.info("Source Condition map object:{}",sourceCondnMap);

        String finalSrcCondition =  returnSrcConditionObject(sourceCondnMap.keySet(),sourceEntity);
        LOGGER.info("finalSourceCondition:{}",finalSrcCondition);
        if(!finalSrcCondition.isEmpty())
        {
            LOGGER.info("going inside the final source condition");
            LOGGER.info("Final source condition Object:{}",finalSrcCondition);
            LOGGER.info("Target Entity:{}",targetEntity);
            executeStatus(cfg,sourceEntity,sourceCondnMap.get(finalSrcCondition),targetEntity);
        }

    }

    public Map<String,List<StatusMapObject>> returnStatusMapObjectList() throws IOException {
        ObjectMapper objectMapper = new ObjectMapper();
        File file = new File(statusFilePath);
        Map<String,List<StatusMapObject>> sourceCondnMap = new HashMap<>();

        List<StatusMapObject> statusMapObjects = objectMapper
                .readValue(file, new TypeReference<List<StatusMapObject>>() {});

        //storing the status info data into map
        for(StatusMapObject mapObject: statusMapObjects)
        {
            sourceCondnMap
                    .computeIfAbsent(mapObject.getSourceCondition(),
                            map->new ArrayList<>()).add(mapObject);
        }

        return sourceCondnMap;

    }

    public String returnSrcConditionObject(Set<String> srcCondns, EntityType sourceEntity)
    {
        for(String srcCond: srcCondns)
        {
            if(verifyCondition(srcCond,sourceEntity))
            {
                LOGGER.info("verified source Condition:{}",srcCond);
                return srcCond;
            }
        }
        return "";
    }

    public boolean verifyCondition(String srcCond,EntityType sourceEntity)
    {
        String[] srcFieldsValues =  srcCond.split(",");
        for(String fieldValues: srcFieldsValues )
        {
            String field = fieldValues.split(":")[0];
            String value = fieldValues.split(":")[1];

            String sourceValue = sourceEntity
                    .getAttribute()
                    .stream()
                    .filter(a->a.getName().equals(field))
                    .findFirst()
                    .get()
                    .getValue()
                    .get(0)
                    .getValue();

            if(!sourceValue.equals(value))
            {
                return false;
            }
        }
        return true;
    }

    public void executeStatus(Jira2JiraCfg cfg,EntityType sourceEntity,List<StatusMapObject> statusMapObjectList, EntityType targetEntity) throws JiraException, ParseException {
        String targetObjectStatus = targetEntity.getAttribute()
                .stream()
                .filter(a->a.getName().equals("status"))
                .findFirst()
                .get()
                .getValue()
                .get(0)
                .getValue();


        String msg = "";

        int Index = returnIndexForTargetMatch(statusMapObjectList,targetEntity);
        LOGGER.info("status object index:{}",Index);
        if(Index!= -1)
        {

            boolean isStatusTimeStampLatest = compareTimeStamps(statusMapObjectList.get(Index),sourceEntity,targetEntity);

            LOGGER.info("timeStampValue:{}",isStatusTimeStampLatest);

            for(int i=Index;i<statusMapObjectList.size();i++)
            {
                StatusMapObject mapObject = statusMapObjectList.get(i);

                EntityType execEntity = new EntityType();

                String[] targetMandatoryFieldValArr = mapObject.getMandatoryFieldsValues().split(",");
                LOGGER.info("targetMandatoryFieldValArr:{}", Arrays.toString(targetMandatoryFieldValArr));
                for(String mandFieldVal:targetMandatoryFieldValArr)
                {
                    String[] mandFieldValArr = mandFieldVal.split(":");

                    LOGGER.info("mandFieldValArr:{}", Arrays.toString(mandFieldValArr));

                    if(mandFieldValArr.length>0 && !mandFieldValArr[0].isEmpty())
                    {
                        if(mandFieldValArr[0].equals("assignee"))
                        {
                            //handle assignee
                            String targetAssigneeEmail = "";
                            try{
                                 targetAssigneeEmail = targetEntity
                                        .getAttribute()
                                        .stream()
                                        .filter(a->a.getName().equals("assigneeDisplayName"))
                                        .findFirst()
                                        .get()
                                        .getValue()
                                        .get(0)
                                        .getValue();
                            }catch (Exception e)
                            {
                                e.printStackTrace();
                            }


                            if(targetAssigneeEmail.isEmpty() || targetAssigneeEmail == null)
                            {
                                execEntity.getAttribute().add(new AttributeType(mandFieldValArr[0], DataTypes.TEXT,new ValueType(mandFieldValArr[1])));
                            }

                        }else{

                            String targetFieldVal = targetEntity
                                    .getAttribute()
                                    .stream()
                                    .filter(a->a.getName().equals(mandFieldValArr[0]))
                                    .findFirst()
                                    .get()
                                    .getValue()
                                    .get(0)
                                    .getValue();



                            if(targetFieldVal.isEmpty())
                            {
                                //special condition for duedate
                                if(mandFieldValArr[0].equals("duedate"))
                                {
                                    execEntity.getAttribute().add(new AttributeType(mandFieldValArr[0], DataTypes.TEXT,new ValueType(getDueDate())));
                                }else{
                                    execEntity.getAttribute().add(new AttributeType(mandFieldValArr[0], DataTypes.TEXT,new ValueType(mandFieldValArr[1])));
                                }

                            }

                        }
                    }


                }

                String targetId = targetEntity
                        .getAttribute()
                        .stream()
                        .filter(a->a.getName().equals("key"))
                        .findFirst()
                        .get()
                        .getValue()
                        .get(0)
                        .getValue();


                //execute transiition
                if(isStatusTimeStampLatest)
                {
                    if(mapObject.getCommentText().isEmpty())
                    {
                        doTransition(cfg,targetId, mapObject.getTransitionId(), execEntity);
                    }else{
                        doTransition(cfg,targetId, mapObject.getTransitionId(), execEntity,mapObject.getCommentText());
                    }

                }


                targetObjectStatus = mapObject.getIntermediateState();

                if(mapObject.getIntermediateState().equals(mapObject.getTargetFinalState()) && mapObject.getTargetFinalState().equals(targetObjectStatus))
                {
                    break;
                }

            }
        }

        //analysis
    }

    public int returnIndexForTargetMatch(List<StatusMapObject> statusMapObjectList,EntityType targetEntity)
    {
        for(int i=0;i<statusMapObjectList.size();i++)
        {
            if(verifyCondition(statusMapObjectList.get(i).getTargetCondition(),targetEntity))
            {
                return i;
            }
        }
        return -1;
    }

    public boolean compareTimeStamps(StatusMapObject mapObject,EntityType sourceEntity, EntityType targetEntity) throws com.hazelcast.internal.json.ParseException, java.text.ParseException
    {
        if(!mapObject.isIncludeStatusTimeStampCondn()) return true;

        String srcLastStatusStamp = sourceEntity
                .getAttribute()
                .stream()
                .filter(a->a.getName().equals(mapObject.getSourceTimeStampField()))
                .findFirst()
                .get()
                .getValue()
                .get(0)
                .getValue();

        String destLastStatusStamp = targetEntity
                .getAttribute()
                .stream()
                .filter(a->a.getName().equals(mapObject.getTargetTimeStampField()))
                .findFirst()
                .get()
                .getValue()
                .get(0)
                .getValue();

        srcLastStatusStamp = srcLastStatusStamp.substring(0,srcLastStatusStamp.length()-5);
        destLastStatusStamp = destLastStatusStamp.substring(0,destLastStatusStamp.length()-5);

        SimpleDateFormat sdFormat = new SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss");
        Date srcDate = sdFormat.parse(srcLastStatusStamp);
        Date destDate = sdFormat.parse(destLastStatusStamp);

        return srcDate.after(destDate);
    }


    private void doTransition(Jira2JiraCfg cfg,String targetId, String transitionId, EntityType entity) throws JiraException
    {
        LOGGER.info("inside the status exchange doTransition method");
        LOGGER.info("target ID:{}",targetId);
        LOGGER.info("transition ID:{}",transitionId);
        LOGGER.info("Entity:{}",entity);


        DoTransitionInput input = new DoTransitionInput();
        input.setConfigSetName(cfg.destConfigSet());
        input.setIssueKey(targetId);
        input.setTransitionId(transitionId);
        input.setEntity(entity);

        jira.doTransition(input);
    }


    private void doTransition(Jira2JiraCfg cfg,String targetId, String transitionId, EntityType entity,String commentText) throws JiraException
    {
        LOGGER.info("inside the status exchange doTransition comment method");
        LOGGER.info("target ID:{}",targetId);
        LOGGER.info("transition ID:{}",transitionId);
        LOGGER.info("Entity:{}",entity);


        DoTransitionInput input = new DoTransitionInput();
        input.setConfigSetName(cfg.destConfigSet());
        input.setIssueKey(targetId);
        input.setTransitionId(transitionId);
        input.setEntity(entity);
        input.setComment(commentText);

        jira.doTransition(input);
    }


    private String getDueDate()
    {
        ZonedDateTime now = ZonedDateTime.now();
        ZonedDateTime futureDate = now.plusDays(5);

        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ss");

        return futureDate.format(formatter);
    }
}
