import requests
import json


class LLMService:

    def __init__(self):
        self.url = "http://localhost:11434/api/generate"
        self.model = "llama3:8b"

    def ask(self, prompt: str, temperature: float = 0.2):

        response = requests.post(
            self.url,
            json={
                "model": self.model,
                "prompt": prompt,
                "temperature": temperature,
                "stream": False
            },
            timeout=60
        )

        data = response.json()
        return data.get("response", "").strip()
