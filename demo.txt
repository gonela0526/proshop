package org.sumanth.process;

import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.sumanth.lib.*;

import java.io.File;
import java.io.IOException;
import java.text.ParseException;
import java.text.SimpleDateFormat;
import java.time.ZonedDateTime;
import java.time.format.DateTimeFormatter;
import java.util.*;

public class StatusExchange {

    private static final Logger LOGGER = LoggerFactory.getLogger(StatusExchange.class);


    private final String statusFilePath;
    private final EntityType sourceEntity;
    private final EntityType targetEntity;
    private final Jira2JiraCfg cfg;
    private final JiraOSGiServiceI jira;

    public StatusExchange(String statusFilePath, EntityType sourceEntity, EntityType targetEntity, Jira2JiraCfg cfg, JiraOSGiServiceI jira) {
        this.statusFilePath = statusFilePath;
        this.sourceEntity = sourceEntity;
        this.targetEntity = targetEntity;
        this.cfg = cfg;
        this.jira = jira;
    }

    public void execute() throws IOException, JiraException, ParseException {
        LOGGER.info("Executing Status Exchange processing engine.");

        ObjectMapper objectMapper = new ObjectMapper();
        List<StatusRule> rules = objectMapper.readValue(new File(statusFilePath), new TypeReference<List<StatusRule>>() {});

        // Find the single rule that matches the Source Entity state properties
        StatusRule matchingRule = rules.stream()
                .filter(rule -> verifyConditions(rule.getSourceConditions(), sourceEntity))
                .findFirst()
                .orElse(null);

        if (matchingRule == null) {
            LOGGER.warn("No status mapping rule found matching the source entity context.");
            return;
        }

        executeTransitions(matchingRule);
    }

    private void executeTransitions(StatusRule rule) throws JiraException, ParseException {
        List<TransitionStep> steps = rule.getTransitions();

        // Find where to start in our transition array sequence based on current target status
        int startIndex = -1;
        for (int i = 0; i < steps.size(); i++) {
            if (verifyConditions(steps.get(i).getTargetConditions(), targetEntity)) {
                startIndex = i;
                break;
            }
        }

        if (startIndex == -1) {
            LOGGER.warn("Target entity does not match any current starting transition state configurations.");
            return;
        }

        // Validate timestamp condition if required before progressing
        if (!isTimestampValid(rule.getTimestampCondition())) {
            LOGGER.info("Skipping transition execution: Source timestamp is older than Target timestamp.");
            return;
        }

        String targetId = getEntityAttributeValue(targetEntity, "key").orElseThrow(() -> new JiraException("Target Key is missing"));

        // Iterate sequentially through steps starting from matched state position
        for (int i = startIndex; i < steps.size(); i++) {
            TransitionStep step = steps.get(i);
            EntityType execEntity = new EntityType();

            // Populate mandatory parameters safely dynamically
            if (step.getMandatoryFields() != null) {
                for (Map.Entry<String, String> fieldEntry : step.getMandatoryFields().entrySet()) {
                    String fieldName = fieldEntry.getKey();
                    String defaultValue = fieldEntry.getValue();

                    if ("assignee".equalsIgnoreCase(fieldName)) {
                        String currentAssignee = getEntityAttributeValue(targetEntity, "assigneeDisplayName").orElse("");
                        if (currentAssignee.isEmpty()) {
                            execEntity.getAttribute().add(new AttributeType(fieldName, DataTypes.TEXT, new ValueType(defaultValue)));
                        }
                    } else {
                        String currentVal = getEntityAttributeValue(targetEntity, fieldName).orElse("");
                        if (currentVal.isEmpty()) {
                            String assignedValue = "duedate".equalsIgnoreCase(fieldName) ? getDueDate() : defaultValue;
                            execEntity.getAttribute().add(new AttributeType(fieldName, DataTypes.TEXT, new ValueType(assignedValue)));
                        }
                    }
                }
            }

            // Perform State Mutation via Transition execution
            if (step.getCommentText() == null || step.getCommentText().isEmpty()) {
                doTransition(targetId, step.getTransitionId(), execEntity);
            } else {
                doTransition(targetId, step.getTransitionId(), execEntity, step.getCommentText());
            }

            // Stop cascading transitions if target final state target objective has been met
            if (step.getIntermediateState().equals(rule.getTargetFinalState())) {
                LOGGER.info("Reached target final milestone: {}", rule.getTargetFinalState());
                break;
            }
        }
    }

    private boolean verifyConditions(Map<String, String> conditions, EntityType entity) {
        if (conditions == null || conditions.isEmpty()) return false;

        for (Map.Entry<String, String> condition : conditions.entrySet()) {
            String entityValue = getEntityAttributeValue(entity, condition.getKey()).orElse("");
            if (!entityValue.equals(condition.getValue())) {
                return false;
            }
        }
        return true;
    }

    private boolean isTimestampValid(TimestampCondition tsCond) throws ParseException {
        if (tsCond == null || !tsCond.isInclude()) return true;

        String srcStamp = getEntityAttributeValue(sourceEntity, tsCond.getSourceField()).orElse("");
        String targetStamp = getEntityAttributeValue(targetEntity, tsCond.getTargetField()).orElse("");

        if (srcStamp.isEmpty() || targetStamp.isEmpty()) return false;

        // Clean trailing offsets safely
        srcStamp = srcStamp.substring(0, Math.min(srcStamp.length(), 19));
        targetStamp = targetStamp.substring(0, Math.min(targetStamp.length(), 19));

        SimpleDateFormat sdFormat = new SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss");
        Date srcDate = sdFormat.parse(srcStamp);
        Date targetDate = sdFormat.parse(targetStamp);

        return srcDate.after(targetDate);
    }

    private Optional<String> getEntityAttributeValue(EntityType entity, String attributeName) {
        return entity.getAttribute().stream()
                .filter(a -> a.getName().equalsIgnoreCase(attributeName))
                .findFirst()
                .flatMap(a -> a.getValue().stream().findFirst())
                .map(ValueType::getValue);
    }

    private void doTransition(String targetId, String transitionId, EntityType entity) throws JiraException {
        doTransition(targetId, transitionId, entity, null);
    }

    private void doTransition(String targetId, String transitionId, EntityType entity, String comment) throws JiraException {
        LOGGER.info("Executing transition ID {} for Issue Key {}", transitionId, targetId);
        DoTransitionInput input = new DoTransitionInput();
        input.setConfigSetName(cfg.destConfigSet());
        input.setIssueKey(targetId);
        input.setTransitionId(transitionId);
        input.setEntity(entity);
        if (comment != null) {
            input.setComment(comment);
        }
        EntityType output = jira.doTransition(input);
        updateTargetEntity(output,targetEntity);

    }

    private String getDueDate() {
        return ZonedDateTime.now().plusDays(5).format(DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ss"));
    }

    public void updateTargetEntity(EntityType output, EntityType targetEntity) {
        if (output == null || targetEntity == null || output.getAttribute() == null || targetEntity.getAttribute() == null) {
            return;
        }

        output.getAttribute().stream()
                .forEach(a -> {
                    targetEntity.getAttribute().stream()
                            .filter(ta -> ta.getName().equals(a.getName()))
                            .findFirst()
                            .ifPresent(matchedTargetAttribute -> {
                                // Update type
                                matchedTargetAttribute.setType(a.getType());

                                // Deep copy or assign the values list
                                if (a.getValue() != null) {
                                    matchedTargetAttribute.setValue(new ArrayList<>(a.getValue()));
                                } else {
                                    matchedTargetAttribute.setValue(new ArrayList<>());
                                }
                            });
                });
    }

}















package org.sumanth.process;

import java.util.List;
import java.util.Map;

public class StatusRule {
    private String descriptionLabel;
    private Map<String, String> sourceConditions;
    private String targetFinalState;
    private List<TransitionStep> transitions;
    private TimestampCondition timestampCondition;

    public String getDescriptionLabel() {return descriptionLabel;}

    public void setDescriptionLabel(String descriptionLabel) {this.descriptionLabel = descriptionLabel;}

    public Map<String, String> getSourceConditions() {return sourceConditions;}

    public void setSourceConditions(Map<String, String> sourceConditions) {this.sourceConditions = sourceConditions;}

    public String getTargetFinalState() {return targetFinalState;}

    public void setTargetFinalState(String targetFinalState) {this.targetFinalState = targetFinalState;}

    public List<TransitionStep> getTransitions() {return transitions;}

    public void setTransitions(List<TransitionStep> transitions) {this.transitions = transitions;}

    public TimestampCondition getTimestampCondition() {return timestampCondition;}

    public void setTimestampCondition(TimestampCondition timestampCondition) {this.timestampCondition = timestampCondition;}
}





package org.sumanth.process;

public class TimestampCondition {
    private boolean include;
    private String sourceField;
    private String targetField;

    public boolean isInclude() {return include;}

    public void setInclude(boolean include) {this.include = include;}

    public String getSourceField() {return sourceField;}

    public void setSourceField(String sourceField) {this.sourceField = sourceField;}

    public String getTargetField() {return targetField;}

    public void setTargetField(String targetField) {this.targetField = targetField;}
}





package org.sumanth.process;

import java.util.Map;

public class TransitionStep {

    private Map<String,String> targetConditions;
    private String intermediateState;
    private String transitionId;
    private Map<String,String> mandatoryFields;
    private String commentText;

    public Map<String, String> getTargetConditions() {return targetConditions;}

    public void setTargetConditions(Map<String, String> targetConditions) {this.targetConditions = targetConditions;}

    public String getIntermediateState() {return intermediateState;}

    public void setIntermediateState(String intermediateState) {this.intermediateState = intermediateState;}

    public String getTransitionId() {return transitionId;}

    public void setTransitionId(String transitionId) {this.transitionId = transitionId;}

    public Map<String, String> getMandatoryFields() {return mandatoryFields;}

    public void setMandatoryFields(Map<String, String> mandatoryFields) {this.mandatoryFields = mandatoryFields;}

    public String getCommentText() {return commentText;}

    public void setCommentText(String commentText) {this.commentText = commentText;}
}
