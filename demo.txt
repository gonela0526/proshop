import requests
import json


class LLMService:

    def __init__(self):
        self.url = "http://localhost:11434/api/chat"
        self.model = "llama3:8b"   # change if needed

    def classify_intent(self, query: str):

        prompt = f"""
You are an intent classifier.

Return JSON only.

Possible intents:
schedule_query
config_check
filter_query
mapping_query
defect_history_query
failure_summary_query
top_failure_reason_query
global_health_query
unknown

Query:
{query}
"""

        response = self._call_llm(prompt)

        try:
            return json.loads(response)
        except:
            return {"intent": "unknown"}

    def enhance_response(self, user_query: str, system_response: dict):

        prompt = f"""
User asked:
{user_query}

System generated:
{json.dumps(system_response)}

If user is asking for:
- timezone conversion
- explanation
- filtering
- summarization
- transformation

Return improved response in clean markdown.
Otherwise return EXACT original JSON.
"""

        response = self._call_llm(prompt)

        try:
            return json.loads(response)
        except:
            return {"ai_response": response}

    def _call_llm(self, prompt: str):

        payload = {
            "model": self.model,
            "messages": [
                {"role": "user", "content": prompt}
            ]
        }

        res = requests.post(self.url, json=payload)
        data = res.json()

        return data["message"]["content"]
