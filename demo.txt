def classify_intent(self, query: str):

    prompt = f"""
You are an intent classifier for a backend API system.

You must respond with STRICT JSON only.
No markdown.
No explanation.
No extra text.

Your task:
Detect if the user wants to modify or transform the PREVIOUS response.

Return ONLY one of these intents:

1. "transform_previous"
   -> if the user wants to:
      - convert timezone
      - convert format
      - rephrase
      - summarize
      - simplify
      - explain previous result
      - change units
      - modify previously shown data
      - refer to "above", "previous", "that result"

2. "none"
   -> if the query is a new independent question.

User Query:
"{query}"

Return format:
{{ "intent": "<one of the above>" }}
"""

    response = self._call_llm(prompt)

    try:
        return json.loads(response)
    except:
        return None
